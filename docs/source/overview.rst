.. _overview-section:
Overview
=========================

Figures of every UTDQuake network (:ref:`overview-seismic-data`, :ref:`overview-DAS-data`). Search by
network, catalog, country or continent; click a figure to open it full size, or **Open folder** for the
full collection.

.. tip::

   These figures can be created for any network with the plot functions of :class:`utdquake.Network`, or
   your own plots with the :mod:`utdquake.utils.plot` module.

   <script>
   (function () {
     var cards = Array.prototype.slice.call(document.querySelectorAll(".utdq-card"));
     var search = document.getElementById("utdq-search");
     var continent = document.getElementById("utdq-continent");
     var version = document.getElementById("utdq-version");
     var count = document.getElementById("utdq-count");
     function fill(select, attr) {
       var values = {};
       cards.forEach(function (c) { if (c.dataset[attr]) { values[c.dataset[attr]] = true; } });
       Object.keys(values).sort().forEach(function (v) {
         var o = document.createElement("option"); o.value = v; o.textContent = v; select.appendChild(o);
       });
     }
     fill(continent, "continent");
     fill(version, "version");
     function update() {
       var words = search.value.toLowerCase().split(/\s+/).filter(Boolean);
       var shown = 0;
       cards.forEach(function (c) {
         var ok = words.every(function (w) { return c.dataset.search.indexOf(w) !== -1; })
           && (!continent.value || c.dataset.continent === continent.value)
           && (!version.value || c.dataset.version === version.value);
         c.style.display = ok ? "" : "none";
         if (ok) { shown++; }
       });
       count.textContent = "Showing " + shown + " of " + cards.length + " networks";
     }
     [search, continent, version].forEach(function (el) {
       el.addEventListener("input", update); el.addEventListener("change", update);
     });
     update();
   })();
   </script>
